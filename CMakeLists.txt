cmake_minimum_required(VERSION 3.24)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT DEFINED CMAKE_CUDA_COMPILER)
  find_program(GRAPHWEFT_NVCC nvcc HINTS "$ENV{CONDA_PREFIX}/bin" "$ENV{HOME}/.conda/envs/torch2.8/bin")
  if(GRAPHWEFT_NVCC)
    set(CMAKE_CUDA_COMPILER "${GRAPHWEFT_NVCC}" CACHE FILEPATH "CUDA compiler")
  endif()
endif()
if(NOT CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 70 CACHE STRING "CUDA architectures" FORCE)
endif()
project(GraphWeft LANGUAGES CXX CUDA VERSION 0.1.0)
include(FetchContent)
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(spdlog GIT_REPOSITORY https://github.com/gabime/spdlog.git GIT_TAG 6fa36017cfd5731d617e1a934f0e5ea9c4445b13 GIT_SHALLOW FALSE)
FetchContent_Declare(gflags GIT_REPOSITORY https://github.com/gflags/gflags.git GIT_TAG ae8ec2b1b2238ecdd40bb84f1a2e65b1e2c85b00 GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(spdlog gflags)
find_package(CUDAToolkit REQUIRED)
add_library(graphweft_kernels STATIC src/kernels.cu)
target_include_directories(graphweft_kernels PUBLIC include)
target_link_libraries(graphweft_kernels PUBLIC CUDA::cudart)
set_target_properties(graphweft_kernels PROPERTIES CXX_STANDARD 17 CUDA_STANDARD 17 CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
add_library(graphweft STATIC src/graph.cpp src/engine.cpp src/checkpoint.cpp src/scheduler.cpp src/iteration_model.cpp)
target_include_directories(graphweft PUBLIC include)
target_link_libraries(graphweft PUBLIC graphweft_kernels spdlog::spdlog gflags::gflags)
target_compile_features(graphweft PUBLIC cxx_std_17)
add_executable(graphweft_cli app/main.cpp)
target_link_libraries(graphweft_cli PRIVATE graphweft)
add_executable(graphweft_validate tests/validate.cpp)
target_link_libraries(graphweft_validate PRIVATE graphweft)
add_executable(graphweft_frontier_validate tests/frontier.cu)
target_link_libraries(graphweft_frontier_validate PRIVATE graphweft_kernels)
set_target_properties(graphweft_frontier_validate PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
add_executable(graphweft_kernel_lab kernel_lab/replay.cpp)
target_link_libraries(graphweft_kernel_lab PRIVATE graphweft)
add_executable(graphweft_push_rounds kernel_lab/push_rounds.cpp)
target_link_libraries(graphweft_push_rounds PRIVATE graphweft)
add_executable(graphweft_pull_rounds kernel_lab/pull_rounds.cpp)
target_link_libraries(graphweft_pull_rounds PRIVATE graphweft)
add_executable(graphweft_capture_pull_rounds kernel_lab/capture_pull_rounds.cpp)
target_link_libraries(graphweft_capture_pull_rounds PRIVATE graphweft)
add_executable(graphweft_profile_pull kernel_lab/profile_pull.cpp)
target_link_libraries(graphweft_profile_pull PRIVATE graphweft)
add_executable(graphweft_degree_pull kernel_lab/degree_adaptive_pull.cu)
target_link_libraries(graphweft_degree_pull PRIVATE graphweft)
set_target_properties(graphweft_degree_pull PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
add_executable(graphweft_road_thread_pull kernel_lab/road_thread_pull.cu)
target_link_libraries(graphweft_road_thread_pull PRIVATE graphweft)
set_target_properties(graphweft_road_thread_pull PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
add_executable(graphweft_pull_qgroup kernel_lab/pull_qgroup.cu)
target_link_libraries(graphweft_pull_qgroup PRIVATE graphweft)
set_target_properties(graphweft_pull_qgroup PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
add_executable(graphweft_pull_wide_qgroup kernel_lab/pull_wide_qgroup.cu)
target_link_libraries(graphweft_pull_wide_qgroup PRIVATE graphweft)
set_target_properties(graphweft_pull_wide_qgroup PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
add_executable(graphweft_pull_smem_shuffle kernel_lab/pull_smem_shuffle.cu)
target_link_libraries(graphweft_pull_smem_shuffle PRIVATE graphweft)
set_target_properties(graphweft_pull_smem_shuffle PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
add_executable(graphweft_pull_wide_smem_shuffle kernel_lab/pull_wide_smem_shuffle.cu)
target_link_libraries(graphweft_pull_wide_smem_shuffle PRIVATE graphweft)
set_target_properties(graphweft_pull_wide_smem_shuffle PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
add_executable(graphweft_strided_query_clear kernel_lab/strided_query_clear.cu)
target_link_libraries(graphweft_strided_query_clear PRIVATE graphweft)
set_target_properties(graphweft_strided_query_clear PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
add_executable(graphweft_partition_validate tests/push_partitions.cpp)
target_link_libraries(graphweft_partition_validate PRIVATE graphweft)
add_executable(graphweft_pull_partition_validate tests/pull_partitions.cpp)
target_link_libraries(graphweft_pull_partition_validate PRIVATE graphweft)
